408static inline size_t GetGpioPinsByCategory(HfPinCategory category, HfFunctionalGpioPin* output, size_t max_count) {
409 size_t count = 0;
410 for (size_t i = 0; i < HF_GPIO_MAPPING_SIZE && count < max_count; ++i) {
411 if (HfFunctionalGpioPinCategories[i] == category) {
412 output[count++] = static_cast<HfFunctionalGpioPin>(i);
413 }
414 }
415 return count;
416}

425static inline size_t GetGpioPinsByChipType(HfGpioChipType chip_type, HfFunctionalGpioPin* output, size_t max_count) {
426 size_t count = 0;
427 for (size_t i = 0; i < HF_GPIO_MAPPING_SIZE && count < max_count; ++i) {
428 if (HF_GPIO_MAPPING[i].chip_type == static_cast<uint8_t>(chip_type)) {
429 output[count++] = static_cast<HfFunctionalGpioPin>(i);
430 }
431 }
432 return count;
